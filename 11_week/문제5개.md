[문제5]
{1. 현재 위치 구하기 }
enum MOVE {
    Go,
    Back,
    Left,
    Right,
};

int main() {
    int x = 0, y = 0; 
    char cmd[5] = {Go, Right, Go, Back, Right};
    
    for (int i = 0; i<5; i++) {
        enum MOVE move = cmd[i];
        switch (move) {
            case Go:
                y--;
                break;
            case Back:
                y++;
                break;
            case Left:
                x--;
                break;
            case Right:
                x++;
                break;
            default:
                x = 0, y = 0;
                break;
        }
    }
    
    printf("현재 위치 [%d , %d]", x, y);
}
1. (1, -1)
2. (2, -1)
3. (2, 1)
4. (1, 2)
